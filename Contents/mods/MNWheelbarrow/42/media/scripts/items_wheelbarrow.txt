module MNWheelbarrow
{
    imports { Base }

    /*
     * FASE A0 - SPIKE.
     *
     * Objetivo unico: descobrir se um item container aguenta
     * RequiresEquippedBothHands. Nenhum container do jogo base faz isso
     * (51 ocorrencias em weapon.txt, 21 em normal.txt, 1 em moveable.txt,
     * ZERO em container.txt), entao isto nao tem precedente no vanilla.
     *
     * Icon, WorldStaticModel e os modelos de mao sao PLACEHOLDERS emprestados
     * do Toolbox e da Suitcase. Serao trocados quando o modelo 3D existir.
     *
     * Capacity tem teto de engine de 50 menos o peso do proprio item
     * (50 - 8 = 42). O valor real de uso e elevado em runtime por Lua na
     * Fase C; aqui fica so o suficiente para testar.
     */
    item Wheelbarrow
    {
        DisplayCategory = Container,
        ItemType = base:container,
        Weight = 8.0,

        Icon = Toolbox,
        WorldStaticModel = ToolBox_Ground,
        ReplaceInPrimaryHand = Bag_Suitcase_RHand holdingbagright,
        ReplaceInSecondHand = Bag_Suitcase_LHand holdingbagleft,

        Capacity = 40,
        WeightReduction = 0,
        CanHaveHoles = false,

        RunSpeedModifier = 0.80,
        RequiresEquippedBothHands = TRUE,

        OpenSound = OpenBag,
        CloseSound = CloseBag,
        PutInSound = PutItemInBag,

        Tags = MNWheelbarrow:hauler,
    }
}
