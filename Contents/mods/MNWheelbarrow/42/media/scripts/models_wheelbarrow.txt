module Base
{
    /*
     * Modelo do carrinho quando largado no chao.
     *
     * POR QUE module Base E NAO module MNWheelbarrow:
     * todos os modelos do jogo base vivem em `module Base` -- models_items.txt
     * inteiro e Base -- e nenhum item vanilla referencia modelo qualificado por
     * modulo. Definir o modelo num modulo proprio e referencia-lo pelo nome nu
     * e um caminho sem nenhum precedente no jogo, e o modelo simplesmente nao
     * aparecia no mundo, sem erro no log. O nome carrega o prefixo MNWheelbarrow
     * para nao colidir com outro mod dentro do namespace compartilhado.
     *
     * O mesh original tinha 4 texturas PBR de 4096x4096 embutidas (21.7 MB).
     * O PZ so usa mapa difuso, entao o payload embutido foi removido; a
     * geometria ficou identica (2240 vertices, mesmos 393 nos).
     *
     * SCALE: comparar so a bounding box do mesh NAO funciona. O tamanho final e
     *
     *     bbox do mesh  x  Lcl Scaling do no Model  x  scale deste script
     *
     * e o Lcl Scaling difere brutalmente entre os arquivos:
     *
     *     Log vanilla   bbox 30.932  x  Lcl 0.02778  x  0.4  =  0.3437
     *     este mesh     bbox  1.900  x  Lcl 100.0            <- 100x embutido
     *
     * O vanilla embute a conversao de unidade no Lcl Scaling (91.44/2.54 = 36,
     * e 1/36 = 0.02778). Este arquivo veio de uma ferramenta em centimetros com
     * a unidade nao aplicada, entao carrega 100x. Ignorar isso me fez calcular
     * scale = 6.5, que renderizou o carrinho 3593x grande demais -- ocupando
     * varios quarteiroes.
     *
     * 0.0057 e o valor calibrado em tela, em tres passadas: 0.0025 ficou tres
     * vezes pequeno demais, 0.0075 ficou 20% grande demais, 0.006 ainda pedia
     * mais 5%. Dobrar o numero dobra o tamanho renderizado.
     *
     * ALTURA: o mesh foi deslocado 0.380 no eixo Y (ver
     * assets/tools_fbx_shift_y.py) para a base ficar em Y = 0 em vez de a
     * origem ficar no centro. Os modelos vanilla sao centrados -- medi o Log e
     * o ToolBox_Ground e ambos tem a origem a 50% da altura -- e o engine
     * assenta a origem no chao, entao todo modelo afunda metade da propria
     * altura. Num tronco deitado isso e imperceptivel; num carrinho alto o
     * resultado era metade dele enterrada.
     */
    model MNWheelbarrow_Ground
    {
        mesh = WorldItems/Wheelbarrow,
        texture = WorldItems/Wheelbarrow,
        scale = 0.0057,
    }

    /*
     * Modelo do carrinho SEGURADO NAS MAOS.
     *
     * Tecnica aprendida do mod Carry Visible Items: um item aparece nas maos
     * de duas formas, e a segunda dispensa rigging --
     *
     *   ReplaceInPrimaryHand  exige mesh COM SKIN preso ao osso, como as bolsas
     *                         do jogo base fazem
     *   StaticModel + AnimMask usa mesh ESTATICO comum, e a pose vem de uma
     *                         mascara de animacao
     *
     * Usamos a segunda. Os modelos de mao do PZ nao tem bloco de offset -- o
     * deslocamento em relacao ao osso da mao esta assado na malha, e por isso os
     * arquivos daquele mod tem sufixo _Hand. Geramos o nosso com
     * assets/tools_fbx_shift.py a partir do MESMO FBX original.
     *
     * GIRO: 180 em X, e este numero e PREVISTO, nao chutado.
     *
     * O espaco do osso da mao inverte o eixo em relacao ao espaco do mundo, e a
     * relacao exata e
     *
     *     aparencia em jogo com giro T  =  aparencia no mundo com giro (180 + T)
     *
     * validada em tools_preview_hand_pose.py contra QUATRO observacoes
     * independentes de tela -- T = 0, 90, -90 e -45 -- todas batendo. Logo o
     * giro que produz a pose natural (a mesma do modelo de chao, que ja esta
     * correta) e T = 180.
     *
     * Isso tambem explica o resultado da rodada 1, que eu havia lido errado:
     * T = 180 apareceu "nas costas como uma mochila". A ORIENTACAO estava certa
     * desde ali; o que estava errado era a translacao, que jogava o carrinho
     * para tras do personagem. Passei tres rodadas girando quando o problema
     * era deslocar.
     *
     * TRANSLACAO: Rx(180) nega Y e Z, entao um dy NEGATIVO sobe no mundo. -0.6
     * ergue o carrinho a altura da mao. O eixo de frente/tras e o Z e continua
     * em calibracao -- ver as variantes abaixo.
     */
    model MNWheelbarrow_Hand
    {
        mesh = WorldItems/Wheelbarrow_Hand,
        texture = WorldItems/Wheelbarrow,
        scale = 0.0057,
    }

    /* Calibracao, rodada 4 -- descartaveis.
       A ROTACAO saiu de calibracao: 180 em X e previsto e validado (ver acima).
       As tres agora usam essa mesma pose e variam so a TRANSLACAO, para achar o
       eixo de frente/tras -- a unica incognita que sobrou.
       O eixo lateral ja e conhecido (X, medido na rodada 2) e fica em zero nas
       tres, senao o resultado de frente/tras vira ambiguo.
         P1  Z = 0      so a altura, para conferir se -0.6 assenta a mao certo
         P2  Z = -0.7   empurra num sentido
         P3  Z = +0.7   empurra no sentido oposto
       Um dos dois sentidos poe o carrinho a frente do personagem, e a diferenca
       entre P2 e P3 diz qual -- inclusive se Z nao e o eixo de frente/tras, caso
       em que P2 e P3 sairao iguais a P1. */
    model MNWheelbarrow_Hand_P1
    {
        mesh = WorldItems/Wheelbarrow_Hand_P1,
        texture = WorldItems/Wheelbarrow,
        scale = 0.0057,
    }
    model MNWheelbarrow_Hand_P2
    {
        mesh = WorldItems/Wheelbarrow_Hand_P2,
        texture = WorldItems/Wheelbarrow,
        scale = 0.0057,
    }
    model MNWheelbarrow_Hand_P3
    {
        mesh = WorldItems/Wheelbarrow_Hand_P3,
        texture = WorldItems/Wheelbarrow,
        scale = 0.0057,
    }
}
