module MNWheelbarrow
{
    imports { Base }

    /*
     * DESCARTAVEL -- uma pergunta so, sai antes de publicar.
     *
     * PERGUNTA: com a animacao 2H_Heavy ligada, a mascara da mao esquerda ajuda
     * ou estorva?
     *
     * A animacao de dois bracos agora vem de WB_Anim, setando a variavel Weapon =
     * "heavy" -- uma animacao de CORPO INTEIRO. As mascaras sao outra camada:
     * remendos por cima da animacao, aplicados por mao. Com a animacao certa
     * embaixo, secondaryAnimMask pode passar a ATRAPALHAR, puxando o braco
     * esquerdo para fora da pose.
     *
     * primaryAnimMask fica nos DOIS itens, e nao por simetria: esta medido neste
     * projeto que StaticModel so renderiza quando existe uma mascara. Sem ela o
     * carrinho desaparece da mao. Entao a mascara primaria nao e ajustavel, e o
     * unico campo que este teste varia e o secundario.
     *
     *   item real     mascara primaria + secundaria
     *   TestOneMask   so a primaria
     *
     * Qual dos dois tiver os bracos mais coerentes com a animacao vence, e o
     * outro campo sai.
     */
    item TestOneMask
    {
        DisplayCategory = Container,
        ItemType = base:container,
        Weight = 3.0,
        Icon = Toolbox,
        Capacity = 20,
        WorldStaticModel = MNWheelbarrow_Ground,
        StaticModel = MNWheelbarrow_Hand,
        primaryAnimMask = holdingbagright,
        RequiresEquippedBothHands = TRUE,
        Tags = MNWheelbarrow:mnwbHauler,
    }
}
