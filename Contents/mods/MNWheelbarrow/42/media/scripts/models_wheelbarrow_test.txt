    /*
     * VARREDURA DE BASE DA ROTACAO DA MAO -- descartavel, sai antes de publicar.
     *
     * POR QUE UMA VARREDURA INTEIRA DE UMA VEZ, E NAO MAIS TRES VARIANTES:
     *
     * A pose que o jogador ve e o produto de duas rotacoes,
     *
     *     aparencia  =  B . R . malha
     *
     * onde R e o que este arquivo controla e B e a matriz do osso Bip01_Prop2,
     * desconhecida. Quatro rodadas foram gastas variando um angulo de Euler por
     * vez e tentando inferir B do resultado. Isso nao converge: rotacoes nao
     * comutam, entao variar X depois de ter mudado Z nao mede a mesma coisa que
     * variar X sozinho, e um B ajustado a quatro observacoes foi refutado pela
     * quinta.
     *
     * Identidade mais as seis rotacoes de 90 graus nos eixos formam uma BASE:
     * observar B.R para esses R determina B por inteiro, sem inferencia e sem
     * ambiguidade de ordem de Euler -- cada uma tem um unico eixo, entao a
     * convencao de ordem nao interfere. Com B na mao, o R da pose desejada sai
     * por conta, de uma vez.
     *
     * Custa uma rodada de teste em vez de N, e o resultado e uma medicao em vez
     * de um ajuste. As oito compartilham a MESMA malha do modelo de chao -- so
     * o rotate muda, entao qualquer diferenca observada e da rotacao.
     */
    model MNWB_RotA { mesh = WorldItems/Wheelbarrow, texture = WorldItems/Wheelbarrow, scale = 0.0057,
        attachment Bip01_Prop2 { offset = 0.0 0.0 0.0, rotate = 0.0 0.0 0.0, } }

    model MNWB_RotB { mesh = WorldItems/Wheelbarrow, texture = WorldItems/Wheelbarrow, scale = 0.0057,
        attachment Bip01_Prop2 { offset = 0.0 0.0 0.0, rotate = 90.0 0.0 0.0, } }

    model MNWB_RotD { mesh = WorldItems/Wheelbarrow, texture = WorldItems/Wheelbarrow, scale = 0.0057,
