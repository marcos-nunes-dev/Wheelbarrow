module Base
{
    /*
     * SPIKE -- o carrinho como VEICULO.
     *
     * POR QUE VEICULO, depois de duas arquiteturas descartadas:
     *
     *   item na mao      movimento suave, teto de capacidade 50
     *   objeto do mundo  teto 100, mas movimento aos saltos de tile
     *   animal (cavalo)  movimento suave, mas o alforje e um ITEM -- teto 50
     *   VEICULO          movimento suave E capacidade ate 1000
     *
     * O veiculo e o unico cruzamento das duas exigencias. E, alem disso, e o
     * unico que da de graca o que o Marcos quer expandir depois: pneu com
     * condicao, furo, remendo e integracao com a pericia de mecanica. Nas
     * outras rotas tudo isso teria que ser inventado em Lua.
     *
     * O QUE ESTE SPIKE PRECISA RESPONDER, em ordem de risco:
     *   1. um veiculo customizado minimo carrega e spawna?
     *   2. a capacidade do porta-malas passa mesmo de 50?
     *   3. como o personagem aparece na tela "dirigindo" isto?
     *
     * A resposta 3 decide se vale investir em animacao customizada -- que e
     * possivel (o mod de cavalos faz, com AnimSets/player e anims_X proprios) e
     * e a parte mais pesada daquele mod.
     *
     * RESPOSTAS JA OBTIDAS no primeiro teste: o veiculo carrega e spawna, o
     * modelo 3D renderiza, e a capacidade do porta-malas chegou a 64 -- acima
     * do teto de 50 dos itens. A rota entrega o que prometia.
     */
    vehicle MNWheelbarrow
    {
        mechanicType = 1,
        engineRepairLevel = 1,
        playerDamageProtection = 0.0,
        neverSpawnKey = true,

        model
        {
            file = Vehicles_MNWheelbarrow,
            scale = 1.0,
            offset = 0.0 0.0 0.0,
        }
        skin
        {
            texture = Vehicles/wheelbarrow,
        }

        /* O modelo saiu BRANCO no primeiro teste. O shader "vehicle" nao usa so
           a textura difusa: ele espera um conjunto de mapas, e o trailer vanilla
           declara sete. Sem a mascara o shader nao tem o que multiplicar.

           A nossa mascara e totalmente transparente de proposito -- ela marca
           zonas de dano e ferrugem por cor, e um carrinho nao precisa dessa
           divisao. As demais apontam para as texturas compartilhadas do jogo,
           que e o que o proprio trailer faz. */
        textureMask = Vehicles/wheelbarrow_mask,
        textureDamage1Overlay = Vehicles/Veh_Blood_Mask,
        textureDamage2Overlay = Vehicles/Veh_Blood_Hvy,
        textureDamage1Shell = Vehicles/Veh_Damage1,
        textureDamage2Shell = Vehicles/Veh_Damage2,
        textureRust = Vehicles/Veh_Rust,

        /* Numeros MEDIDOS na malha, nao estimados. Com scale 0.0088 o modelo
           renderiza 0.785 x 0.669 x 1.672, entao as meias-extensoes sao: */
        extents = 0.392 0.334 0.836,
        mass = 40,
        physicsChassisShape = 0.392 0.334 0.836,
        centerOfMassOffset = 0.0 -0.20 0.30,
        shadowExtents = 0.392 0.836,
        shadowOffset = 0.0 0.0,

        /* Um carrinho e empurrado por pessoa: forca baixa e velocidade de
           caminhada, nao de veiculo. */
        engineForce = 400,
        engineQuality = 100,
        engineLoudness = 0,
        maxSpeed = 8.0,
        brakingForce = 20,
        stoppingMovementForce = 3.0,
        rollInfluence = 1.0,
        steeringIncrement = 0.08,
        steeringClamp = 0.6,
        suspensionStiffness = 60.0,
        suspensionCompression = 2.83,
        suspensionDamping = 2.88,
        maxSuspensionTravelCm = 4.0,
        suspensionRestLength = 0.1,
        wheelFriction = 3.0,
        frontEndHealth = 60,
        rearEndHealth = 60,
        seats = 1,

        physics box
        {
            offset = 0.0 0.0 0.0,
            extents = 0.392 0.30 0.836,
            rotate = 0.0 0.0 0.0,
        }

        /* A roda foi MEDIDA na malha: 884 vertices no extremo +Z com |x| < 0.12,
           ocupando y de 0 a 0.600 e z de 0.552 a 0.950. Convertido para a escala
           renderizada, isso da raio 0.264 e centro em z 0.661, y -0.070.

           Os "pes" de tras entram como rodas minusculas porque a fisica de
           veiculo precisa de mais de um ponto de apoio -- com uma roda so o
           corpo tomba. Elas ficam na altura em que o modelo encosta no chao. */
        wheel FrontCenter
        {
            front = true,
            offset = 0.0 -0.070 0.661,
            radius = 0.264,
            width = 0.08,
        }
        wheel RearLeft
        {
            front = false,
            offset = 0.20 -0.274 -0.55,
            radius = 0.06,
            width = 0.05,
        }
        wheel RearRight
        {
            front = false,
            offset = -0.20 -0.274 -0.55,
            radius = 0.06,
            width = 0.05,
        }

        /* Os nomes TruckBed e SeatFrontLeft nao sao escolha estetica: os
           templates Trunk e Seat definem pecas com esses nomes exatos, e as
           areas precisam casar. Inventar nomes proprios quebraria o template. */
        area TruckBed
        {
            xywh = 0.0 0.0 0.8 0.9,
        }
        /* A area do assento e onde o jogador FICA EM PE para entrar, e por isso
           precisa cair FORA do corpo. O veiculo tem meia-profundidade 0.836,
           entao qualquer coisa acima de -0.836 esta dentro dele -- o -0.75 que
           eu tinha posto ficava embutido no carrinho, e o pathfinding nao tem
           para onde levar o personagem. Agora fica atras, no espaco livre. */
        area SeatFrontLeft
        {
            xywh = 0.0 -1.15 0.7 0.7,
        }

        template = Trunk/part/TruckBed,

        /* O template Seat NAO e usado, mas o itemType E necessario.
           
           O template traz uma regra coringa "part Seat*" com tabela de
           instalacao que exige um banco de carro em inventario -- exigencia sem
           sentido para um carrinho. Ao remove-la, porem, eu tirei junto o
           itemType, e o diagnostico mostrou o custo disso:

               peca TruckBed        instalada=true   (tem itemType)
               peca SeatFrontLeft   instalada=false  (sem itemType)
               getBestSeat = -1

           Uma peca so conta como instalada quando tem um item associado. Sem
           itemType nao existe o que instalar, e o assento nunca fica pronto --
           dai "nao tem assento instalado" e o E sem efeito.

           Com o itemType e sem a tabela de instalacao, a peca nasce instalada
           como a caçamba, sem exigir que o jogador ache um banco. */
        part SeatFrontLeft
        {
            category = seat,
            area = SeatFrontLeft,
            itemType = Base.NormalCarSeat,
            container
            {
                seat = FrontLeft,
            }
            /* SEM ISTO A PECA NASCE VAZIA.
               itemType diz QUAL item a peca aceita; este bloco e que manda
               CRIAR esse item quando o veiculo aparece. Vehicles.Create.Default
               chama VehicleUtils.createPartInventoryItem(part), e sem essa
               chamada getInventoryItem() fica nil, a peca conta como nao
               instalada e getBestSeat() devolve -1.

               Foi a diferenca que o diagnostico expos entre as duas pecas: a
               caçamba vem do template Trunk, que ja traz este bloco, e nascia
               instalada; o assento, declarado a mao, nao. */
            lua
            {
                create = Vehicles.Create.Default,
            }
        }

        /* SEM ISTO NAO DA PARA ENTRAR NO VEICULO. O template Seat cria a peca,
           mas quem define onde o jogador fica e por onde ele entra e o bloco
           passenger -- eu tinha omitido.

           "inside" fica ATRAS do carrinho, em z negativo, que e onde alguem se
           posiciona para empurrar. E o mais perto que se chega da pose certa
           sem animacao customizada. */
        passenger FrontLeft
        {
            /* ESTES BLOCOS anim SAO OBRIGATORIOS. Sem eles a acao de entrar
               entra na fila e nao acontece nada -- exatamente o sintoma que
               tivemos. O jogo base os define no template PassengerCommon, com
               uma regra coringa "passenger *", e eu nao os havia replicado.

               anim idle e tambem o gancho para o personagem PARECER que empurra
               o carrinho: e so trocar o nome da animacao aqui. Por enquanto usa
               a animacao neutra, ate escolhermos uma que tenha os bracos a
               frente. */
            area = SeatFrontLeft,
            /* hasRoof controla se o jogo ESCONDE quem esta dentro. Com true o
               personagem sumia ao entrar, como num carro fechado. Um carrinho
               de mao nao tem teto, e queremos ver o personagem empurrando, entao
               false.

               (Eu tinha trocado para true suspeitando que fosse o que travava a
               entrada. Nao era: o problema era a UI de assentos nao registrar o
               clique, e a entrada funciona pela tecla numerica.) */
            hasRoof = false,
            anim enter
            {
                anim = Climb_WindowA,
                rate = 0.33,
            }
            anim idle
            {
                anim = Idle,
                rate = 1.0,
            }
            anim exit
            {
                anim = Climb_WindowB,
                rate = 0.23,
            }
            position inside
            {
                offset = 0.0 -0.10 -0.75,
                rotate = 0.0 0.0 0.0,
            }
            /* Altura junto ao chao, como no carro de luxo vanilla, cuja posicao
               externa fica em y = -0.41. Eu tinha usado -0.10, meia altura do
               corpo -- o jogador nao fica flutuando no meio do veiculo. */
            position outside
            {
                offset = 0.0 -0.30 -1.15,
                rotate = 0.0 0.0 0.0,
                area = SeatFrontLeft,
            }
        }

        /* A caçamba. E aqui que a capacidade passa de 50: container preso a
           peca de veiculo tem teto de 1000 no engine, contra 50 de container
           preso a item. Containers de veiculo do jogo base chegam a 160, e o
           padrao de sobrescrever a capacidade vem de vehicle_stepvan_template. */
        part TruckBed
        {
            itemType = Base.NormalTrunk,
            container
            {
                capacity = 100,
                /* O template Trunk liga conditionAffectsCapacity, e a
                   capacidade passa a escalar com a condicao da peca -- foi por
                   isso que o teste mostrou 64 em vez de 100. Um carrinho de mao
                   nao tem porta-malas que amassa, entao desligamos. */
                conditionAffectsCapacity = false,
            }
        }
    }
}
