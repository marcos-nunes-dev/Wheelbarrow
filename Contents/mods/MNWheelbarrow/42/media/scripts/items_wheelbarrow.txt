module MNWheelbarrow
{
    imports { Base }

    /*
     * O carrinho de mao.
     *
     * Duas mecanicas nao sao definidas aqui, e sim em Lua (WB_Weight.lua),
     * porque o engine nao consegue expressa-las em script:
     *
     *   Capacity      - o valor real vem de SandboxVars via setCapacity(),
     *                   que aceita qualquer numero em runtime.
     *   WeightReduction - o engine aplica UMA porcentagem sobre o peso total
     *                   do conteudo. Como o carrinho so deve aliviar itens
     *                   pesados, a porcentagem equivalente e recalculada a
     *                   cada mudanca de conteudo via setWeightReduction().
     *
     * Os valores abaixo sao so o ponto de partida antes do Lua assumir.
     *
     * Icon, WorldStaticModel e os modelos de mao sao PLACEHOLDERS emprestados
     * do Toolbox e da Suitcase, ate o modelo 3D proprio existir. Trocar esses
     * campos nao muda o ID do item, entao saves existentes continuam validos.
     */
    item Wheelbarrow
    {
        DisplayCategory = Container,
        ItemType = base:container,

        /* Peso baixo por realismo: um carrinho de mao vazio e leve.
           Ele NAO limita mais a capacidade -- o teto de 50 que essa conta antes
           assumia nao existe. Ver o cabecalho de WB_Weight.lua. */
        Weight = 3.0,

        /* Modelo de chao e de mao proprios (ver models_wheelbarrow.txt).
           So o icone segue placeholder: ele precisa de um render 2D do modelo,
           que e trabalho separado. */
        Icon = Toolbox,
        WorldStaticModel = MNWheelbarrow_Ground,

        /* Aparece nas maos com mesh ESTATICO, sem rigging.
           StaticModel + AnimMask e a tecnica que o mod Carry Visible Items usa
           para o tanque de oxigenio e o extintor. A alternativa,
           ReplaceInPrimaryHand, exigiria mesh com skin preso ao osso -- que e
           trabalho de Blender que eu nao consigo produzir.

           holdingbag deixa os bracos para baixo ao lado do corpo, que e a pose
           mais parecida com segurar os cabos de um carrinho entre as poses que
           o jogo oferece (as outras sao tocha e guarda-chuva). */
        StaticModel = MNWheelbarrow_Hand,

        /* As duas mascaras, uma por braco. Nao existe mascara de duas maos: as
           nove que o jogo base usa sao todas pares por lado (bag, torch,
           umbrella, angletorch), entao a unica forma de por os dois bracos na
           pose e declarar as duas.

           ESTE CAMPO JA FOI REMOVIDO DAQUI, POR DIAGNOSTICO ERRADO MEU. Eu
           concluira que secondaryAnimMask impedia equipar, comparando o item real
           (nao equipava) com uma variante de teste (equipava). Mas havia DUAS
           diferencas entre os dois, nao uma: a mascara, e o WB_Hands, cuja
           invariante estava quebrada e desequipava o carrinho no mesmo frame. A
           variante escapava por nao estar em HAULER_TYPES, entao o WB_Hands nunca
           a tocava. Consertar o WB_Hands e o que resolveu equipar -- a mascara era
           inocente, e eu credite o resultado a variavel errada.

           O jogo base sustenta isso: 12 itens usam StaticModel com as duas
           mascaras (as lanternas Hurricane). Nenhum e container, mas a combinacao
           de campos em si esta longe de ser inedita. */
        primaryAnimMask = holdingbagright,
        secondaryAnimMask = holdingbagleft,

        /* Valor de PARTIDA apenas: WB_Weight sobe para o da sandbox option
           (200 por padrao) assim que o save carrega. Fica baixo aqui porque o
           parser de script e que pode ter limite proprio -- o limite de runtime,
           esse sim, nao existe. */
        Capacity = 47,
        WeightReduction = 0,
        CanHaveHoles = false,

        RunSpeedModifier = 0.80,

        /* RequiresEquippedBothHands: TERCEIRO campo que eu havia removido por
           diagnostico errado, e o terceiro a voltar depois de testado isolado.
           Ele equipa sem problema nenhum.

           O que ele entrega e a UI: ISInventoryPaneContextMenu marca force2Hands
           quando isRequiresEquippedBothHands() e verdadeiro e troca "equipar na
           primaria" / "equipar na secundaria" por uma unica opcao "Equip Two
           Hands", com traducao pronta em todos os idiomas do jogo.

           A causa real de o item nao equipar era o WB_Hands, que o desequipava no
           mesmo frame. Enquanto isso estava quebrado eu culpei tres campos
           diferentes -- secondaryAnimMask, RequiresEquippedBothHands e o modelo --
           porque comparava o item real com variantes que o WB_Hands nao tocava.
           Nenhum dos tres tinha culpa.

           O WB_Hands continua necessario: e ele que impede uma arma na mao livre
           e conserta estados que deixavam o personagem invisivel. */
        RequiresEquippedBothHands = TRUE,

        AcceptItemFunction = MNWheelbarrow.AcceptItem,
        Tooltip = Tooltip_item_MNWheelbarrow,

        OpenSound = OpenBag,
        CloseSound = CloseBag,
        PutInSound = PutItemInBag,

        /* base:heavyitem foi TESTADA e nao entregou: o gerador usa essa tag e e
           carregado com dois bracos, mas no carrinho a animacao continuou de um
           braco so. Removida em vez de mantida "por garantia" -- ela nao serve ao
           que eu a adicionei, e semanticamente contradiz o item (o carrinho pesa
           3.0 justamente para maximizar a capacidade, entao pesado ele nao e) e
           toca exatamente os caminhos de carregar-item-pesado que ja deixaram o
           personagem invisivel neste projeto. */
        Tags = MNWheelbarrow:mnwbHauler,
    }
}
